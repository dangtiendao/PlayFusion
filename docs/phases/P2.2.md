### B. Tóm Tắt DDL & Ý Nghĩa 6 Bảng Lõi

```sql
-- 1. PROFILES (Phase P2.1c)
-- Quản lý danh tính người chơi, liên kết auth.users, bảo vệ role
CREATE TABLE public.profiles (
  user_id uuid PRIMARY KEY REFERENCES auth.users(id) ON DELETE CASCADE,
  display_name text NOT NULL CHECK (char_length(trim(display_name)) >= 2 AND char_length(trim(display_name)) <= 20),
  avatar_url text NULL,
  role text NOT NULL DEFAULT 'player' CHECK (role IN ('player', 'moderator', 'admin')),
  is_anonymous boolean NOT NULL DEFAULT true,
  created_at timestamptz NOT NULL DEFAULT now(),
  updated_at timestamptz NOT NULL DEFAULT now()
);

-- 2. GAMES (Phase P2.2a)
-- Bản chiếu vận hành của Frontend Registry, quản trị bật/tắt game
CREATE TABLE public.games (
  id text PRIMARY KEY,
  name text NOT NULL,
  category text NOT NULL CHECK (category IN ('board', 'arcade', 'puzzle', 'skill', 'party')),
  ranked boolean NOT NULL DEFAULT false,
  rating_system text NOT NULL DEFAULT 'leaderboard_only' CHECK (rating_system IN ('elo', 'glicko2', 'leaderboard_only')),
  scoring text NOT NULL CHECK (scoring IN ('win_loss', 'score', 'time')),
  min_players int NOT NULL CHECK (min_players >= 1),
  max_players int NOT NULL CHECK (max_players >= min_players),
  is_enabled boolean NOT NULL DEFAULT true,
  ranked_enabled boolean NOT NULL DEFAULT true,
  created_at timestamptz NOT NULL DEFAULT now(),
  updated_at timestamptz NOT NULL DEFAULT now()
);

-- 3. SEASONS (Phase P2.2a)
-- Quản lý mùa giải thi đấu, Partial Unique Index đảm bảo duy nhất 1 mùa active
CREATE TABLE public.seasons (
  id smallint GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
  name text NOT NULL,
  started_at timestamptz NOT NULL,
  ended_at timestamptz NULL,
  is_active boolean NOT NULL DEFAULT false,
  created_at timestamptz NOT NULL DEFAULT now()
);
CREATE UNIQUE INDEX idx_seasons_one_active ON public.seasons (is_active) WHERE is_active = true;

-- 4. MATCHES (Phase P2.2b)
-- Lưu trữ bản ghi ván đấu, chuỗi nước đi nén (text), ràng buộc tính nhất quán
CREATE TABLE public.matches (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  game_id text NOT NULL REFERENCES public.games(id),
  mode text NOT NULL CHECK (mode IN ('solo', 'vs_ai', 'local_pvp', 'online_1v1', 'online_ffa', 'online_team')),
  season_id smallint NULL REFERENCES public.seasons(id),
  is_ranked boolean NOT NULL DEFAULT false,
  started_at timestamptz NOT NULL,
  ended_at timestamptz NULL,
  duration_ms int NULL CHECK (duration_ms IS NULL OR duration_ms >= 0),
  end_reason text NULL CHECK (end_reason IS NULL OR end_reason IN ('normal', 'resign', 'timeout', 'disconnect', 'abort')),
  engine_options text NULL,
  final_state text NULL,
  moves text NULL,
  created_at timestamptz NOT NULL DEFAULT now(),
  CONSTRAINT chk_matches_ended_consistency CHECK ((ended_at IS NULL) = (end_reason IS NULL))
);

-- 5. MATCH_PARTICIPANTS (Phase P2.2b - Thiết Kế Dọc)
-- Lưu từng đấu thủ/bot theo ghế, chống 1 user ngồi 2 ghế
CREATE TABLE public.match_participants (
  match_id uuid NOT NULL REFERENCES public.matches(id) ON DELETE CASCADE,
  seat_index smallint NOT NULL CHECK (seat_index >= 0),
  user_id uuid NULL REFERENCES public.profiles(user_id),
  is_bot boolean NOT NULL DEFAULT false,
  bot_level text NULL CHECK (bot_level IS NULL OR bot_level IN ('easy', 'medium', 'hard')),
  result text NULL CHECK (result IS NULL OR result IN ('win', 'loss', 'draw')),
  placement smallint NULL CHECK (placement IS NULL OR placement >= 1),
  score numeric NULL,
  rating_before int NULL, rating_after int NULL, rating_delta int NULL,
  rd_before numeric NULL, rd_after numeric NULL,
  PRIMARY KEY (match_id, seat_index),
  CONSTRAINT chk_participant_bot_user CHECK ((is_bot = true AND user_id IS NULL) OR (is_bot = false)),
  CONSTRAINT chk_participant_bot_level CHECK ((is_bot = true) = (bot_level IS NOT NULL))
);
CREATE UNIQUE INDEX idx_match_participants_unique_user ON public.match_participants (match_id, user_id) WHERE user_id IS NOT NULL;

-- 6. PLAYER_RATINGS (Phase P2.2c)
-- Bảng điểm danh vọng theo mùa, bộ đếm tăng dần, sẵn sàng Glicko-2
CREATE TABLE public.player_ratings (
  user_id uuid NOT NULL REFERENCES public.profiles(user_id) ON DELETE CASCADE,
  game_id text NOT NULL REFERENCES public.games(id),
  season_id smallint NOT NULL REFERENCES public.seasons(id),
  rating numeric NOT NULL DEFAULT 1200,
  rd numeric NOT NULL DEFAULT 350,
  volatility numeric NOT NULL DEFAULT 0.06,
  games_played int NOT NULL DEFAULT 0 CHECK (games_played >= 0),
  wins int NOT NULL DEFAULT 0 CHECK (wins >= 0),
  losses int NOT NULL DEFAULT 0 CHECK (losses >= 0),
  draws int NOT NULL DEFAULT 0 CHECK (draws >= 0),
  streak int NOT NULL DEFAULT 0,
  best_rating numeric NOT NULL DEFAULT 1200,
  placement_done boolean NOT NULL DEFAULT false,
  last_played_at timestamptz NULL,
  created_at timestamptz NOT NULL DEFAULT now(),
  updated_at timestamptz NOT NULL DEFAULT now(),
  PRIMARY KEY (user_id, game_id, season_id),
  CONSTRAINT chk_player_ratings_matches_count CHECK (wins + losses + draws <= games_played)
);
```

Sau khi chạy xong trên PROD, hãy dán câu lệnh sau vào SQL Editor trên cả 2 môi trường để đối chiếu:

```sql
-- 1. ĐỐI CHIẾU 6 BẢNG LÕI TỒN TẠI (Kỳ vọng: 6 bảng)
SELECT table_name
FROM information_schema.tables
WHERE table_schema = 'public'
  AND table_name IN ('profiles', 'games', 'seasons', 'matches', 'match_participants', 'player_ratings')
ORDER BY table_name;

-- 2. ĐỐI CHIẾU 7 POLICIES RLS TRÊN 6 BẢNG (Kỳ vọng: 7 policies, 100% SELECT + 1 UPDATE profiles)
SELECT tablename, policyname, cmd, roles
FROM pg_policies
WHERE schemaname = 'public'
ORDER BY tablename, policyname;

-- 3. KIỂM TRA DỮ LIỆU SEED KHỞI TẠO (Kỳ vọng: 1 game caro, 1 season Mùa 1 active)
SELECT id, name, category, ranked FROM public.games;
SELECT id, name, is_active FROM public.seasons;
```

## 7. ĐẦU VÀO GỢI Ý CHO PHASE KẾ TIẾP (PHASE P2.3 — SCHEMA KINH TẾ & QUẢN TRỊ)

Khi bắt đầu **Phase P2.3 (Schema Kinh Tế & Quản Trị Hệ Thống)**, kỹ sư cần nắm các thông tin sau:

1. **Khóa Ngoại Tham Chiếu**:
   - `wallets.user_id`, `user_inventory.user_id`, `user_equipped.user_id` tham chiếu đến `public.profiles(user_id) ON DELETE CASCADE`.
   - `user_inventory.item_id` tham chiếu đến `public.shop_items(id)`.
